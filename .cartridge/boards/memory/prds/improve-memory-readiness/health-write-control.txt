import ast,pathlib,tempfile,subprocess,os,time,hashlib,json,signal
base=pathlib.Path('/tmp/memory-readiness-baseline')
lines=(base/'public-test.log').read_text().splitlines()
line=next(x for x in lines if x.strip().startswith('right: ['))
seed=bytes(ast.literal_eval(line.split('right:',1)[1].strip()))
binary='/Users/feb/dev/cartridge/memory.ctg/target/prd-writer-boundary/debug/memory'
from http.server import ThreadingHTTPServer,BaseHTTPRequestHandler
import threading
calls=[]
class Handler(BaseHTTPRequestHandler):
 def log_message(self,*args): pass
 def do_POST(self):
  self.rfile.read(int(self.headers.get('Content-Length','0')));calls.append(self.path)
  time.sleep(.08)
  body=json.dumps({'message':{'role':'assistant','content':'Existing fixture relationship'},'choices':[{'message':{'content':'Existing fixture relationship'}}],'embeddings':[[1.,0.,0.,0.]]}).encode()

server=ThreadingHTTPServer(('127.0.0.1',0),Handler)
threading.Thread(target=server.serve_forever,daemon=True).start()
results=[]
for run in range(6):
 with tempfile.TemporaryDirectory(prefix='mhctrl-',dir='/tmp') as tmp:

   try:
    until=time.monotonic()+10
    while not list(rt.glob('memory-*.sock')) and time.monotonic()<until and p.poll() is None: time.sleep(.001)

   finally:
    
    if p.poll() is None: os.killpg(p.pid,signal.SIGKILL)
